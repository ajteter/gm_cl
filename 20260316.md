## Step 1: Update `src/services/AdService.js` (Social Bar Injection)
We must dynamically inject the Adsterra Social Bar script right before the closing `</body>` tag when a revive is requested. We will use a time-based hack to resolve the promise.

Replace the `AdService.js` content with:
```javascript
export const AdService = {
  // onProgress is an optional callback to update the UI countdown
  showRewardedAd: async (onProgress) => {
    return new Promise((resolve) => {
      console.log("[AdService] Injecting Adsterra Social Bar...");

      // 1. Inject Adsterra Script
      const script = document.createElement('script');
      script.type = 'text/javascript';
      script.src = '[https://pl28926955.effectivegatecpm.com/43/fd/aa/43fdaa575cf88d81705f695c821fe4b7.js](https://pl28926955.effectivegatecpm.com/43/fd/aa/43fdaa575cf88d81705f695c821fe4b7.js)';
      // CRITICAL: Bypass Cloudflare Rocket Loader
      script.setAttribute('data-cfasync', 'false'); 
      
      // Adsterra recommends placing it right above closing </body> tag
      document.body.appendChild(script);

      // 2. Start Hacky Countdown (e.g., 6 seconds)
      let timeLeft = 6;
      const interval = setInterval(() => {
        timeLeft -= 1;
        if (onProgress) onProgress(timeLeft);

        if (timeLeft <= 0) {
          clearInterval(interval);
          console.log("[AdService] Reward time met, granting revive.");
          resolve(true); // Grant reward
        }
      }, 1000);
    });
  }
};

```
